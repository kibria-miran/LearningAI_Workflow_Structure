DAY 1: TRACKER.PY — LOG EVERY CALL
Estimated: 60–90 min | Cost: <$0.005
----------------------------------------------------------------

CONCEPT:
  tracker.py wraps every API call and writes a record to
  results/usage.jsonl (one JSON object per line, one per call).

WHY JSONL?
  Plain text. One JSON object per line. Open in any editor,
  grep it, parse with Python, import into Excel. No setup.

WHY PROMPT HASH?
  You don't store full prompts in logs — they could contain
  sensitive data. The sha256 hash lets you identify "this call
  used the same prompt as that one" without storing the text.
  It also lets you detect prompt drift: if the hash changes
  between runs, your prompt changed.

PERFORMANCE NOTE:
  get_spent() re-reads usage.jsonl on every API call.
  At learning scale (< 200 calls) this is fine.
  For 1000+ calls, cache the running total in memory instead.

CODE:
  # tracker.py
  import os, json, time, hashlib
  from datetime import datetime, timezone
  from dotenv import load_dotenv
  from openai import OpenAI, RateLimitError

  load_dotenv()

  client = OpenAI(
      api_key=os.getenv("DEEPSEEK_API_KEY"),
      base_url="https://api.deepseek.com",
      timeout=30.0
  )
  LOG_FILE = "results/usage.jsonl"
  os.makedirs("results", exist_ok=True)

  # Budget caps — tightened for Phase 3 scale (<$0.30 total expected)
  # Scale these up when you move to production workloads.
  DAILY_BUDGET_USD   = 0.10
  MONTHLY_BUDGET_USD = 2.00

  def _load_prices() -> dict:
      with open("config/prices.json") as f:
          return json.load(f)

  def _calculate_cost(model: str, input_tokens: int, output_tokens: int) -> float:
      prices = _load_prices()
      if model not in prices:
          return 0.0
      p = prices[model]
      return (input_tokens  * p["input_per_million"]  / 1_000_000 +
              output_tokens * p["output_per_million"] / 1_000_000)

  def get_spent(period: str = "daily") -> float:
      """Sum costs from usage.jsonl for today (local time) or this month."""
      if not os.path.exists(LOG_FILE):
          return 0.0
      # Use local time so "daily" aligns with your working day, not UTC.
      now = datetime.now()
      total = 0.0
      with open(LOG_FILE) as f:
          for line in f:
              line = line.strip()
              if not line:
                  continue
              entry = json.loads(line)
              # Log timestamps are UTC ISO strings; convert to local for comparison
              ts = datetime.fromisoformat(entry["timestamp"]).astimezone().replace(tzinfo=None)
              if period == "daily" and ts.date() == now.date():
                  total += entry.get("cost", 0.0)
              elif period == "monthly" and ts.year == now.year and ts.month == now.month:
                  total += entry.get("cost", 0.0)
      return total

  def tracked_call(messages: list, model: str = "deepseek-chat", tools: list = None, **kwargs) -> dict:
      """
      Make an API call and log the result to results/usage.jsonl.

      Args:
          messages : list of message dicts (role + content)
          model    : model name string
          tools    : optional list of tool definitions
          **kwargs : extra params (temperature, max_tokens, etc.)

      Returns:
          dict with keys:
            success (bool), content (str|None), finish_reason (str|None),
            tool_calls (list|None), message (object|None),
            input_tokens (int), output_tokens (int), cost (float), error (str|None)
      """
      # Budget check BEFORE the API call
      daily = get_spent("daily")
      if daily >= DAILY_BUDGET_USD:
          raise RuntimeError(f"Daily budget cap hit: ${daily:.4f} >= ${DAILY_BUDGET_USD}. "
                             f"Reset cap or wait until tomorrow.")
      monthly = get_spent("monthly")
      if monthly >= MONTHLY_BUDGET_USD:
          raise RuntimeError(f"Monthly budget cap hit: ${monthly:.4f} >= ${MONTHLY_BUDGET_USD}.")

      prompt_hash = hashlib.sha256(json.dumps(messages).encode()).hexdigest()[:8]
      start = time.time()
      success = True
      error = None
      content = None
      finish_reason = None
      tool_calls = None
      message = None
      input_tokens = 0
      output_tokens = 0

      call_kwargs = {"model": model, "messages": messages, **kwargs}
      if tools:
          call_kwargs["tools"] = tools

      for attempt in range(3):
          try:
              r = client.chat.completions.create(**call_kwargs)
              choice = r.choices[0]
              content = choice.message.content
              finish_reason = choice.finish_reason
              tool_calls = choice.message.tool_calls
              message = choice.message
              input_tokens = r.usage.prompt_tokens
              output_tokens = r.usage.completion_tokens
              break
          except RateLimitError:
              if attempt < 2:
                  time.sleep(2 ** attempt)  # 1s, then 2s
              else:
                  success = False
                  error = "RateLimitError: max retries exceeded"
          except Exception as e:
              success = False
              error = str(e)
              break

      latency_ms = int((time.time() - start) * 1000)
      cost = _calculate_cost(model, input_tokens, output_tokens)

      log_entry = {
          "timestamp": datetime.now(timezone.utc).isoformat(),
          "model": model,
          "prompt_hash": prompt_hash,
          "input_tokens": input_tokens,
          "output_tokens": output_tokens,
          "cost": cost,
          "latency_ms": latency_ms,
          "success": success,
          "error": error
      }
      with open(LOG_FILE, "a") as f:
          f.write(json.dumps(log_entry) + "\n")

      return {
          "success": success,
          "content": content,
          "finish_reason": finish_reason,
          "tool_calls": tool_calls,
          "message": message,
          "input_tokens": input_tokens,
          "output_tokens": output_tokens,
          "cost": cost,
          "error": error
      }


  if __name__ == "__main__":
      result = tracked_call([{"role": "user", "content": "Say hello."}])
      print("Success:", result["success"])
      print("Content:", result["content"])
      print("Cost:   ", result["cost"])
      print("Check results/usage.jsonl")

DAY 4: SPLIT TESTS FROM EVALS
Estimated: 60 min | Cost: $0
----------------------------------------------------------------

CONCEPT:
  TESTS (assertions.json):
    Programmatic. Pass or Fail. A computer checks these automatically.

  EVALS (evals.json):
    Judgment-based. Requires a human (you) to review weekly.
    Open evals.json once a week, run each prompt manually, read
    the output, and fill in last_reviewed (today's date) and
    last_result (your 1–5 score or pass/fail note).

ANALOGY:
  Tests = spell checker. Automatic, objective, fast.
  Evals = human editor. Subjective, slow, catches what spell
          checkers miss.

NOTE on test_003 (harmful request refusal):
  The forbidden list uses broad terms so a compliant model
  (one that would help) fails the test. A refusing model says
  things like "I can't help with that" — none of the forbidden
  words appear, so it passes. This is the correct behavior.

NOTE on non_empty_response:
  This check only verifies the model produced output, not that
  the output is correct. It's a sanity check, not a quality check.

DAY 5: RUN_TESTS.PY — PASS/FAIL, TOTAL COST, SAVE RESULTS
Estimated: 60–90 min | Cost: ~$0.002
----------------------------------------------------------------

CONCEPT:
  run_tests.py reads assertions.json, runs each test, checks the
  output, prints PASS or FAIL, saves results to a dated file,
  and shows total cost at the end.

  Exit code 1 if any test fails, 0 if all pass. This makes it
  usable in CI: python run_tests.py && git commit -m "..."

CODE:
  # run_tests.py
  import json, os, sys
  from datetime import datetime
  from tracker import tracked_call

  def check(output: str, assertion: dict) -> bool:
      if output is None:
          return False
      t = assertion["check_type"]
      if t == "valid_json":
          try: json.loads(output); return True
          except: return False
      elif t == "valid_json_with_key":
          try: return assertion["key"] in json.loads(output)
          except: return False
      elif t in ("under_n_words", "max_words"):
          return len(output.split()) < assertion["n"]
      elif t == "contains":
          return assertion["text"].lower() in output.lower()
      elif t == "contains_any_of":
          return any(w.lower() in output.lower() for w in assertion["required"])
      elif t == "contains_none_of":
          return not any(w.lower() in output.lower() for w in assertion["forbidden"])
      elif t == "non_empty_response":
          # Checks that the model produced output — not that it's correct.
          return len(output.strip()) > 0
      return False

  with open("tests/assertions.json") as f:
      tests = json.load(f)

  passed = 0
  total_cost = 0.0
  results = []

  for t in tests:
      messages = [
          {"role": "system", "content": t["system"]},
          {"role": "user",   "content": t["prompt"]}
      ]
      result = tracked_call(messages)
      total_cost += result["cost"]

      ok = result["success"] and check(result["content"], t)
      if ok:
          passed += 1
          print(f"PASS  {t['id']}: {t['description']}")
      else:
          print(f"FAIL  {t['id']}: {t['description']}")
          if not result["success"]:
              print(f"      Error: {result['error']}")
          else:
              print(f"      Output: {(result['content'] or '')[:100]}")

      results.append({"id": t["id"], "description": t["description"], "passed": ok})

  print(f"\n{passed}/{len(tests)} passed | Total cost: ${total_cost:.6f}")

  os.makedirs("results", exist_ok=True)
  date_str = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
  with open(f"results/{date_str}.json", "w") as f:
      json.dump({"date": date_str, "passed": passed, "total": len(tests),
                 "total_cost_usd": total_cost, "results": results}, f, indent=2)
  print(f"Results saved to results/{date_str}.json")

  sys.exit(0 if passed == len(tests) else 1)

DAY 6: CHANGE A PROMPT, RE-RUN, COMPARE
Estimated: 45 min | Cost: ~$0.004
----------------------------------------------------------------

CONCEPT:
  Every time you change a prompt, re-run tests and compare the
  result files. This is how you know if a change helped or hurt.

TASKS:
  1. Run tests. Note the result filename (call it BASELINE).
  2. Open assertions.json. Find test_007 (step-by-step).
     Remove "Final answer starts with 'Answer:'" from the system prompt.
  3. Run tests again. Note the new result filename.
  4. Compare: did test_007 go from PASS to FAIL?
  5. Restore the original prompt. Run again. Confirm it's back to PASS.

  test_007 is used here because it checks an exact string ("Answer:").
  Removing that instruction reliably breaks the test — no ambiguity.

HOW TO COMPARE:
  Open both result JSON files. Look at the "results" array.
  Find which test IDs changed from passed:true to passed:false.
